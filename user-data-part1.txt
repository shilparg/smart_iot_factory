#!/usr/bin/env python3
import random, time, logging, json
from datetime import datetime
from prometheus_client import Counter, Histogram, start_http_server

logging.basicConfig(filename='/var/log/iot-simulator.log', level=logging.INFO)

event_counter = Counter('events_total','Total events processed')
anomaly_counter = Counter('anomaly_type_total','Total anomalies by type',['type'])
anomaly_severity = Counter('anomaly_severity_total','Total anomalies by severity',['level'])
temp_spike_hist = Histogram('temperature_spike_c','Distribution of temperature spikes',buckets=[5,10,15,20,25,30,35])

def inject_anomaly(data):
    anomaly_type = random.choices(
        ["none","spike","drift","noise","freeze","combined"],
        [0.7,0.1,0.05,0.1,0.03,0.02]
    )[0]
    if anomaly_type=="spike":
        spike_value=random.uniform(15,35)
        data['temperature_c']+=spike_value
        temp_spike_hist.observe(spike_value)
        anomaly_severity.labels(level="critical").inc()
    elif anomaly_type=="drift":
        data['temperature_c']+=random.uniform(0.1,0.5)
        anomaly_severity.labels(level="warning").inc()
    elif anomaly_type=="noise":
        data['rpm']+=int(random.gauss(0,50))
        anomaly_severity.labels(level="info").inc()
    elif anomaly_type=="freeze":
        anomaly_severity.labels(level="warning").inc()
    elif anomaly_type=="combined":
        data['temperature_c']+=random.uniform(5,15)
        data['rpm']-=random.randint(100,300)
        anomaly_severity.labels(level="critical").inc()
    anomaly_counter.labels(type=anomaly_type).inc()
    return data, anomaly_type

def main():
    start_http_server(8000)
    while True:
        data={"temperature_c":random.uniform(20,25),"rpm":random.randint(1500,2000)}
        event_counter.inc()
        data,atype=inject_anomaly(data)
        logging.info(json.dumps({"timestamp":datetime.utcnow().isoformat(),"anomaly_type":atype,"values":data}))
        time.sleep(2)

if __name__=="__main__":
    main()
PYCODE

groups:
  - name: anomaly-alerts
    interval: 30s
    rules:

        data:
          - refId: A

            model:
